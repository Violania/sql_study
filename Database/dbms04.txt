1)where절(조건절)
-특정한 조건에 맞는 내용을 검색하기위해 사용하는 키워드

--사원테이블에서 급여가 10000이상인 사원들의 정보를 출력
select * 
from employees
where salary >= 10000;
--오라클의 비교연산자
>,>=,<,<=,=,!=(<>)

--월급이 5000이상인 사원의 ,사번 ,이름, 성, 급여를 출력
select employee_id,first_name,last_name,salary
from employees
where salary >= 5000;

--사번이 150을 넘는 (초과) 사원들의 사번 성 입사일을 조회
select employee_id, last_name ,HIRE_DATE
from employees
where employee_id > 150;

--성이 smith인 사원의 정보를 사번 성 급여순으로 출력
select employee_id, last_name, salary
from employees
where last_name = 'Smith';

--이름이 마이클인 사원의 사번 이름 급여 부서번호를 조회
select employee_id, first_name, salary, DEPARTMENT_ID
from employees
where first_name = 'Michael';

--직종이 sa_rep인 사원들의 사번 이름 이메일 직종 급여를 조회
select employee_id, first_name, email, job_id, salary
from employees
where job_id = 'SA_REP';

--입사일이 06년 1월 1일 이후인 사원들의 사번 이름 입사일을 조회
select employee_id, first_name, hire_date
from employees
where hire_date > '01/01/2006';

--보너스를 받지 않는 사원들의 이름 급여 보너스순으로 출력
select first_name, salary, COMMISSION_PCT
from employees
where COMMISSION_PCT is null;

--보너스를 받는 사원들 출력
select first_name, salary, COMMISSION_PCT
from employees
where COMMISSION_PCT is not null;

--상사가 없는 사원들의 사번 이름 급여를 출력
select employee_id, first_name, salary
from employees
where MANAGER_ID is null;

2)다중조건
and(&&),or(||) (문자만 )
--급여가 10000이상 15000미만인 사원들의 사번 이름 급여를 출력
select employee_id, first_name, salary
from employees
where salary >= 10000 and salary < 15000; 

--급여가 5000미만 20000이상인 사원들의 사번 이름 급여 입사일을 출력
select employee_id, first_name, salary, hire_date
from employees
where salary < 5000 or salary >= 20000;

--05년에서 06년사이의 입사한 사원들의 이름 입사일 직종을 출력
select first_name, hire_date, job_id
from employees
where hire_date >= '01/01/2005' and hire_date <= '12/31/2006';

--직종이 sa_man이거나 it_rrog인 사원들의 사번 이름 직종을 조회
select employee_id, first_name, job_id
from employees
where job_id = 'SA_MAN' or job_id = 'IT_PROG';

3-1) in연산자(or)
-주어진 목록에서 찿고자 하는 값이 있는 경우
--in 미사용
select employee_id, first_name, job_id
from employees
where job_id = 'SA_MAN' or job_id = 'IT_PROG' or job_id = 'HR_REP';

--in사용
select employee_id, first_name, job_id
from employees
where job_id IN('SA_MAN','IT_PROG','HR_REP');

--사번이 100 102 104 106 번인 사원들의 모든정보 조회
select *
from employees
where employee_id IN(100, 102, 104, 106);


3-2)between 연산자(and)
-A와 B사이의 값을 조회하고자 할 때
--월급이 4000~8000인 사원들의 사번 이름 급여를 출력
--미사용
select employee_id, first_name, salary
from employees
where salary >= 4000 and salary <= 8000; 

--사용
select employee_id, first_name, salary
from employees
where salary BETWEEN 4000 AND 8000;

--05년도 입사사원들의 사번 이름 직종 입사일을 조회
select employee_id, first_name, job_id, hire_date
from employees
--where hire_date >= '01/01/2005' and hire_date <= '12/31/2005';
where hire_date BETWEEN '01/01/2005' AND '12/31/2005';

--급여가 6000~12000사이이면서 50번 부서에서 근무하는 사원들의 이름 급여 직종을 조회

select first_name, salary, job_id
from employees
where  salary BETWEEN 6000 AND 12000 
and DEPARTMENT_ID = 50; 

--부서번호 80이 아닌 부서에서 근무하는 사원들의 사번 이름 직종 부서번호를 조회
select first_name, salary, job_id
from employees
where DEPARTMENT_ID != 80; 
where not DEPARTMENT_ID = 80; 
where DEPARTMENT_ID NOT IN (80); 

--07년도 이후에 입사한 사원들의 정보를 사번 이름 직종 입사일 순으로 출력
select employee_id, first_name, job_id, hire_date
from employees
where hire_date >= '01/01/2007';

--급여가 2200,3200,5000,6800인 사원들의 정보를 사번, 이름 ,직종, 급여순으로 출력
select employee_id, first_name, job_id, salary
from employees
where salary in (2200,3200,5000,6800);

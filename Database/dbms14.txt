--사원이름이 Michael이고,직종이 mk_man안 사원보더 급여를 많이받는(초과) 사원들의 정보를 사번, 이름 ,직종, 급여 순으로 조회

--문제접근1)
-이름이 마이클이고 직종이 mk_man인 사원의 급여 = 13000
select salary from employees
where first_name = 'Michael' and job_id = 'MK_MAN';

--문제접근2)
13000을 초과하는 급여를 받는 사원들의 정보
select employee_id, first_name, job_id, salary
from employees
where salary > 13000;

위의 두 select문 합치기
select employee_id, first_name, job_id, salary
from employees
where salary >(select salary from employees
where first_name = 'Michael' and job_id = 'MK_MAN');

--회사에서 월급을 가장 많이 받는 사원의 사번, 이름, 월급을 조회***
select employee_id, first_name, salary
from employees
where salary = (select max(salary) from employees);

--150번 사원과 같은 월급을 받는 모든 사원들의 정보를 사번,이름, 급여순으로 출력
select employee_id, first_name, salary
from employees
where salary = (select salary from employees where employee_id = 150);

--Bruce와 같은 부서에서 근무중인 사원들의 사번 ,이름을 출력
select employee_id,first_name
from employees
where department_id = ( select department_id from employees where first_name = 'Bruce');

--111번 사원과 같은 직종이면서 159번 사원의 급여를 초과하는 사원들의 정보를 사번,이름 ,직종,입사일 ,급여순으로 조회
select employee_id, first_name, hire_date, salary
from employees
where job_id = (select job_id from employees where employee_id = 111) 
and salary > (select salary from employees where employee_id = 159);

--직종별 급여의 평균을 출력 
평균급여가 bruce의 급여를 초과하는 경우에만 출력
select job_id, avg(salary) 
from employees 
group by job_id
having avg(salary) > (select salary from employees where first_name = 'Bruce');

select e.first_name, e.salary, r.region_name

select avg(salary)

1)Select구문 : db에 저장되어있는 자원들을 검색할 때 사용 

-계정이 소유하고있는 테이블 목록보기
-select * from tabs;

-사원 테이블 조회하기
select * from EMPLOYEES;

-사원테이블에서 사원들의 이름, 성을 조회
select FIRST_NAME,LAST_NAME from EMPLOYEES;

-부서 테이블의 부서번호,부서명을 조회
SELECT department_id,department_name
From departments;

사원 테이블에서 사번, 이름 ,입사일, 월급순으로 조회

select EMPLOYEE_ID,FIRST_NAME,HIRE_DATE,SALARY 
from employees;

--별칭
select employee_id,first_name AS myname (= first_name myname)
from employees;

--사번 이름 직종을 출력하되 별칭을 한글로 지정해서 조회
select employee_id 사번, first_name 이름, JOB_ID 직종
from employees;

--사번, 이름, 직종, 급여, 보너스,보너스의 실제금액를 조회
select employee_id, first_name, salary, COMMISSION_PCT, salary * commission_pct bonus from employees;

--distinct : 중복을 제거하는 명령어
select distinct job_id from employees;

--사원중 관리자(매니저)로써 활동중인 사원이 몇 명인지 사번출력해보기
select distinct MANAGER_ID from employees;

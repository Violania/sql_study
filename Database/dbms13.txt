--부서명, 도시, 국가, 대륙별 근무인원수를 직원수가 많은순으로 정렬 하고 20명 이상인 대륙만 조회**
select d.department_name, l.city, c.country_name, r.region_name, count(*) EMPLOYEE_COUNT
from departments d, locations l, countries c, regions r, employees e
where d.location_id = l.location_id and l.country_id = c.country_id and c.region_id = r.region_id and e.department_id = d.department_id
group by d.department_name,l.city,c.country_name,r.region_name
having count(*) >= 20
order by count(*) DESC;

select d.department_name, l.city, c.country_name, r.region_name, count(*) EMPLOYEE_COUNT
from departments d
join locations l ON d.location_id = l.location_id
join countries c ON l.country_id = c.country_id 
join regions r ON c.region_id = r.region_id
join employees e ON e.department_id = d.department_id
group by d.department_name,l.city,c.country_name,r.region_name
having count(*) >= 20
order by count(*) DESC;

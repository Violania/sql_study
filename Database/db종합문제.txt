--사원테이블의 전체 사원의 평균 임금보다 많은(초과) 사원의 사원번호, 이름, 부서명, 입사일, 급여를 급여오름차순으로 출력
select e.employee_id, e.first_name, d.department_name, e.hire_date, e.salary
from employees e
join departments d ON e.department_id = d.department_id
where salary > (select avg(salary) from employees)
order by salary;

--EMPLOYEES 테이블에서 급여를 많이 받는 순서대로 조회했을 때 6번째부터 10 번째까지 5명의 last_name, first_name, salary를 조회**

select last_name, first_name, salary
 from ( 
	select RANK() OVER( order by rank DESC ) rank, e.*
	from employees e
	)
where rank between 6 and 10;

--각 업무(job) 별로 급여(salary)의 총합을 구하고자 한다. 
-- 급여 총합이 가장 높은 업무부터 업무명(job_title)과 급여총합을 조회하시오. 
-- 단, 급여총합이 30,000보다 큰(초과) 업무만 출력하시오. 

select j.job_title, sum(e.salary)
from employees e
join jobs j ON e.job_id = j.job_id
group by j.job_title
having sum(e.salary) > 30000
order by sum(e.salary) DESC;

--각 사원(employee)에 대해서 사번(employee_id), 이름(first_name), 업무명(job_title), 부서 명(department_name)을 조회
--단, 도시명(city)이 'Seattle'인 지역(location)의 부서 (department)에 근무하는 직원을 사원번호 오름차순으로 출력
select e.emlpoyee_id, e.first_name, j.job_title, d.department_name
from employees e
join departments d ON e.department_id = d.department_id
join jobs j ON e.job_id = j.job_id
join locations l ON d.location_id = l.location_id
where l.city = 'Seattle'
order by employee_id;

--'Sales' 부서에 속한 직원의 이름(first_name), 
--급여(salary), 부서이름(department_name)을 조회하시오. 
--단, 급여는 100번 부서의 평균보다 적게 받는(미만) 직원 정보만 출력되어야 한다. 
select e.salary, d.department_name
from employees e
join departments d ON e.department_id = d.department_id

--사원테이블에서 각 직종(job_id)별 인원과 월급의 평균을 구해서 
--월급 평균의 내림차순으로 정렬하시오
--단, 인원이 3명 이상인 직종만 출력할 것!
select count(j.job_id),avg(e.salary) from employees e
join departments d ON e.department_id = d.department_id
join jobs j ON e.job_id = j.job_id
where count(j.job_id) >= 3
order by sum(e.salary) DESC;

select e.employee_id, e.first_name, e. salary, r.region_name
from employees e
join departments d ON e.department_id = d.department_id
join locations l ON d.location_id = l.location_id
join countries c ON l.country_id = c.country_id 
join regions r ON c.region_id = r.region_id
where r.region_name = 'Europe'
and e.salary > ( 
	select avg(e2.salary)
	from employees e2
	join departments d2 ON e2.department_id = d2.department_id
	join locations l2 ON d2.location_id = l2.location_id
	join countries c2 ON l2.country_id = c2.country_id 
	join regions r2 ON c2.region_id = r2.region_id
	where r2.region_name = 'Americas')

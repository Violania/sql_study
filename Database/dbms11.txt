2)02~06입사자수를 연도별로 조회
select substr(hire_date,-4) || '년도' hire_year, count(*) emp_cnt
from employees
where substr(hire_date,-4) BETWEEN '2001' and '2006'
group by substr(hire_date,-4)
order by hire_year;
